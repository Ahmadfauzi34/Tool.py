def scan_directory(path='.', max_depth=2):
    # Surgical directory metadata scanner
    structure = []
    base_depth = path.count(os.sep)
    for root, dirs, files in os.walk(path):
        curr_depth = root.count(os.sep)
        if curr_depth - base_depth <= max_depth:
            structure.append(dict(path=root, dirs=len(dirs), files=len(files)))
    return structure
